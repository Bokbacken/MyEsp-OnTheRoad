# MyEsp-OnTheRoad - ESP-IDF project. Boards: ES3C35P (esp32s3), Waveshare P4 4.3" (esp32p4).
cmake_minimum_required(VERSION 3.22)
set(PROJECT_VER "0.23.21")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# Managed components differ per target (ESP-Hosted on P4), so keep one lock file per target.
if(NOT IDF_TARGET)
  message(FATAL_ERROR "IDF_TARGET is not set - run `idf.py set-target esp32s3|esp32p4` first")
endif()
idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/dependencies.lock.${IDF_TARGET}")
message(STATUS "Dependencies lock: dependencies.lock.${IDF_TARGET}")
project(MyEsp_OnTheRoad)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(IDF_TARGET STREQUAL "esp32s3")
  # Backport Espressif a16c003 in the build tree; never modify the installed SDK.
  set(spi_fix_tool "${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_idf_spi_fix.py")
  set(spi_original "$ENV{IDF_PATH}/components/esp_hw_support/spi_bus_lock.c")
  set(spi_fixed "${CMAKE_CURRENT_BINARY_DIR}/spi_bus_lock_fixed.c")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${spi_fix_tool}" "${spi_original}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${spi_fix_tool}" "${spi_original}" "${spi_fixed}"
    RESULT_VARIABLE spi_fix_result)
  if(NOT spi_fix_result EQUAL 0)
    message(FATAL_ERROR "Could not prepare verified S3 SPI bus-lock fix")
  endif()
  idf_component_get_property(hw_support_lib esp_hw_support COMPONENT_LIB)
  get_target_property(hw_support_sources ${hw_support_lib} SOURCES)
  set(spi_source_count 0)
  set(hw_support_fixed_sources "")
  foreach(hw_source IN LISTS hw_support_sources)
    get_filename_component(hw_source_name "${hw_source}" NAME)
    if(hw_source_name STREQUAL "spi_bus_lock.c")
      list(APPEND hw_support_fixed_sources "${spi_fixed}")
      math(EXPR spi_source_count "${spi_source_count} + 1")
    else()
      list(APPEND hw_support_fixed_sources "${hw_source}")
    endif()
  endforeach()
  if(NOT spi_source_count EQUAL 1)
    message(FATAL_ERROR "Expected exactly one SPI bus-lock source in esp_hw_support")
  endif()
  set_property(TARGET ${hw_support_lib} PROPERTY SOURCES "${hw_support_fixed_sources}")
endif()
file(GLOB web_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/main/web/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  ${web_sources} "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_web.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_web.py"
  RESULT_VARIABLE web_result)
if(NOT web_result EQUAL 0)
  message(FATAL_ERROR "Could not embed web assets")
endif()

# LVGL builtin heap (CONFIG_LV_MEM_SIZE) -> PSRAM .bss (needs CONFIG_SPIRAM_ALLOW_BSS_SEG_EXTERNAL_MEMORY).
idf_component_get_property(lvgl_lib lvgl__lvgl COMPONENT_LIB)
target_compile_definitions(${lvgl_lib} PRIVATE "LV_ATTRIBUTE_LARGE_RAM_ARRAY=__attribute__((section(\".ext_ram.bss\")))")
